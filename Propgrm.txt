# Programs from the Document

## Program 1: Valid Arithmetic Expression using LEX and YACC

### LEX File (arithmetic.l)
```lex
%{
#include "y.tab.h"
%}

%%
[0-9]+          {return NUMBER;}
[a-zA-Z][a-zA-Z0-9]* {return ID;}
[ \t]           ; /* ignore whitespace */
\n              {return 0;}
.               {return yytext[0];}
%%

int yywrap() {
    return 1;
}
```

### YACC File (arithmetic.y)
```yacc
%{
#include <stdio.h>
#include <stdlib.h>
int yylex();
int yyerror(const char *s);
%}

%token NUMBER ID
%left '+'
%left '*'
%left '/'
%left '-'

%%
exp : exp '+' exp
    | exp '-' exp
    | exp '*' exp
    | exp '/' exp
    | '(' exp ')'
    | NUMBER
    | ID
    ;
%%

int main()
{
    printf("Enter the expression: ");
    if (yyparse() == 0) {
        printf("Valid Expression!\n");
    }
    return 0;
}

int yyerror(const char *s) {
    printf("Invalid Expression!\n");
    exit(1);
}
```

## Program 2: Valid Variable Name using LEX and YACC

### LEX File (variable.l)
```lex
%{
#include "y.tab.h"
%}

%%
[0-9]           { return DIGIT; }
[a-zA-Z]        { return ALPHA; }
\n              { return 0; }
.               { return yytext[0]; }
%%

int yywrap() {
    return 1;
}
```

### YACC File (variable.y)
```yacc
%{
#include <stdio.h>
#include <stdlib.h>
%}

%token DIGIT ALPHA

%%
var : ALPHA
    | var ALPHA
    | var DIGIT
    ;
%%

int main(int argc, char *argv[]) {
    printf("Enter a variable name: ");
    yyparse();
    printf("Valid Variable!\n");
    return 0;
}

int yyerror() {
    printf("Invalid Variable!\n");
    exit(1);
}

int yywrap() {
    return 1;
}
```

## Program 3: Calculator using LEX and YACC

### LEX File (calc.l)
```lex
%{
#include "y.tab.h"
#include <math.h>
%}

%%
[0-9]+          { yylval = atoi(yytext); return NUMBER; }
[0-9]+\.[0-9]+  { yylval = atof(yytext); return NUMBER; }
[+]             { return PLUS; }
[-]             { return MINUS; }
[*]             { return MULTIPLY; }
[/]             { return DIVIDE; }
[(]             { return LPAREN; }
[)]             { return RPAREN; }
[ \t]           ; /* ignore whitespace */
\n              { return 0; }
.               { return yytext[0]; }
%%

int yywrap() {
    return 1;
}
```

### YACC File (calc.y)
```yacc
%{
#include <stdio.h>
#include <stdlib.h>
int yylex();
void yyerror(const char *s);
%}

%union {
    double dval;
}

%token <dval> NUMBER
%token PLUS MINUS MULTIPLY DIVIDE LPAREN RPAREN

%type <dval> expression

%left PLUS MINUS
%left MULTIPLY DIVIDE

%%
calculation : expression { printf("Result: %g\n", $1); }
            ;

expression : expression PLUS expression { $$ = $1 + $3; }
           | expression MINUS expression { $$ = $1 - $3; }
           | expression MULTIPLY expression { $$ = $1 * $3; }
           | expression DIVIDE expression { 
               if($3 == 0) {
                   yyerror("Division by zero!");
               }
               $$ = $1 / $3; 
             }
           | LPAREN expression RPAREN { $$ = $2; }
           | NUMBER { $$ = $1; }
           ;
%%

int main() {
    printf("Simple Calculator\n");
    printf("Enter expression: ");
    yyparse();
    return 0;
}

void yyerror(const char *s) {
    fprintf(stderr, "Error: %s\n", s);
}

int yywrap() {
    return 1;
}
```
